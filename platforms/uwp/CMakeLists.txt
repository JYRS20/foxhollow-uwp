cmake_minimum_required(VERSION 3.25)
set(CMAKE_SYSTEM_NAME WindowsStore)
set(CMAKE_SYSTEM_VERSION 10.0)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
project(foxhollow_uwp_smoke LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if (NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "x64")
endif ()
include(Dependencies.cmake)
include(Runtime.cmake)
add_executable(foxhollow-uwp-smoke WIN32 main.cpp storage.c settings.c smoke.cpp)
target_compile_definitions(foxhollow-uwp-smoke PRIVATE WINAPI_FAMILY=WINAPI_FAMILY_APP)
target_link_libraries(foxhollow-uwp-smoke PRIVATE SDL3::SDL3 LIBUWP dawn::webgpu_dawn WindowsApp)
include(Package.cmake)
foxhollow_uwp_package(foxhollow-uwp-smoke)
